library(dplyr)
library(ggplot2)
library(tidyr)
library(scales)

# Load grade data
data_raw <- read.csv("grades-10years.csv", stringsAsFactors = FALSE)

# Load division mappings from login-info.csv
login_info <- read.csv("login-info.csv", stringsAsFactors = FALSE, check.names = FALSE)
division_map <- login_info %>%
  filter(Rubric != "" & !is.na(Rubric) & Division != "" & !is.na(Division)) %>%
  select(Rubric, Division) %>%
  distinct()

# Expand rubrics (some have multiple comma-separated)
# Take first occurrence of each rubric to avoid duplicates
division_expanded <- division_map %>%
  mutate(Rubric = strsplit(Rubric, ",\\s*")) %>%
  unnest(Rubric) %>%
  mutate(Rubric = trimws(Rubric)) %>%
  distinct(Rubric, .keep_all = TRUE)

# Parse and prepare data
data <- data_raw %>%
  mutate(
    Year = as.integer(substr(TERM, 1, 4)),
    Semester_Code = as.integer(substr(TERM, 5, 6)),
    Semester = case_when(
      Semester_Code == 10 ~ "Winter",
      Semester_Code == 20 ~ "Spring",
      Semester_Code == 60 ~ "Summer",
      Semester_Code == 90 ~ "Fall",
      TRUE ~ "Unknown"
    ),
    Term_Label = paste(Semester, Year),
    Academic_Year = ifelse(Semester == "Fall",
                           paste0(Year, "-", Year + 1),
                           paste0(Year - 1, "-", Year))
  ) %>%
  # Filter out practicum/thesis courses
  mutate(course_num = as.numeric(gsub(".*?(\\d+).*", "\\1", COURSE))) %>%
  filter(!is.na(course_num)) %>%
  filter(!(course_num >= 391 & course_num <= 399)) %>%
  filter(course_num < 491) %>%
  # Keep only letter grades
  filter(GRADE %in% c("A", "A-", "B+", "B", "B-", "C+", "C", "C-", "D", "F")) %>%
  # Join division
  left_join(division_expanded, by = c("SUBJECT" = "Rubric"))

# Define grade order
letter_grades <- c("A", "A-", "B+", "B", "B-", "C+", "C", "C-", "D", "F")
data$GRADE <- factor(data$GRADE, levels = letter_grades)

# Color palette for grades
grade_colors <- c(
  "A" = "#1a5e1a", "A-" = "#2d8e2d",
  "B+" = "#ffd700", "B" = "#ffb300", "B-" = "#ff8c00",
  "C+" = "#e57373", "C" = "#c62828", "C-" = "#8e0000",
  "D" = "#5d4037", "F" = "#212121"
)

Overview

This analysis covers 468,294 undergraduate grade records from Fall 2016 through Fall 2025, excluding practicum, thesis, and other special courses (391-399, 491+).

A/A- Rate by Academic Year

yearly_trend <- data %>%
  filter(Semester %in% c("Fall", "Spring")) %>%  # Focus on regular semesters
  group_by(Academic_Year) %>%
  summarise(
    Total = n(),
    A_Rate = sum(GRADE %in% c("A", "A-")) / n() * 100,
    .groups = "drop"
  ) %>%
  filter(Total > 1000) %>%  # Only years with substantial data
  arrange(Academic_Year)

# Set factor levels in chronological order
yearly_trend$Academic_Year <- factor(yearly_trend$Academic_Year, levels = yearly_trend$Academic_Year)

ggplot(yearly_trend, aes(x = Academic_Year, y = A_Rate, group = 1)) +
  geom_line(color = "#2d8e2d", size = 1.5) +
  geom_point(color = "#1a5e1a", size = 3) +
  geom_text(aes(label = round(A_Rate, 1)), vjust = -1, size = 3.5, fontface = "bold") +
  scale_y_continuous(limits = c(0, 60), breaks = seq(0, 60, 10)) +
  labs(
    title = "A/A- Grade Rate Over Time (Fall & Spring Only)",
    x = "Academic Year",
    y = "Percentage of A/A- Grades"
  ) +
  theme_minimal(base_size = 14) +
  theme(axis.text.x = element_text(angle = 45, hjust = 1))

COVID Impact: Before, During, and After

Code 
covid_periods <- data %>%
  filter(Semester %in% c("Fall", "Spring")) %>%
  mutate(
    Period = case_when(
      Year < 2020 ~ "Pre-COVID\n(2016-2019)",
      Year == 2020 | (Year == 2021 & Semester == "Spring") ~ "COVID\n(Spring 2020-Spring 2021)",
      TRUE ~ "Post-COVID\n(Fall 2021-2025)"
    )
  ) %>%
  group_by(Period, GRADE) %>%
  summarise(Count = n(), .groups = "drop") %>%
  group_by(Period) %>%
  mutate(Percentage = Count / sum(Count) * 100) %>%
  ungroup()

covid_periods$Period <- factor(covid_periods$Period,
                                levels = c("Pre-COVID\n(2016-2019)",
                                          "COVID\n(Spring 2020-Spring 2021)",
                                          "Post-COVID\n(Fall 2021-2025)"))

ggplot(covid_periods, aes(x = GRADE, y = Percentage, fill = Period)) +
  geom_bar(stat = "identity", position = "dodge") +
  scale_fill_manual(values = c("#3498db", "#e74c3c", "#2ecc71")) +
  labs(
    title = "Grade Distribution: Pre-COVID vs COVID vs Post-COVID",
    x = "Grade",
    y = "Percentage",
    fill = "Period"
  ) +
  theme_minimal(base_size = 14) +
  theme(legend.position = "bottom")

Grades by Division

division_data <- data %>%
  filter(!is.na(Division) & Division != "") %>%
  group_by(Division, GRADE) %>%
  summarise(Count = n(), .groups = "drop") %>%
  group_by(Division) %>%
  mutate(Percentage = Count / sum(Count) * 100) %>%
  ungroup()

# Order divisions by A/A- rate
division_order <- division_data %>%
  filter(GRADE %in% c("A", "A-")) %>%
  group_by(Division) %>%
  summarise(A_Rate = sum(Percentage)) %>%
  arrange(desc(A_Rate)) %>%
  pull(Division)

division_data$Division <- factor(division_data$Division, levels = division_order)

ggplot(division_data, aes(x = Division, y = Percentage, fill = GRADE)) +
  geom_bar(stat = "identity", position = "stack") +
  scale_fill_manual(values = grade_colors) +
  labs(
    title = "Grade Distribution by Division",
    subtitle = "Ordered by A/A- rate (highest to lowest)",
    x = "Division",
    y = "Percentage",
    fill = "Grade"
  ) +
  theme_minimal(base_size = 14) +
  theme(legend.position = "right")

division_trends <- data %>%
  filter(!is.na(Division) & Division != "") %>%
  filter(Semester %in% c("Fall", "Spring")) %>%
  group_by(Division, Academic_Year) %>%
  summarise(
    A_Rate = sum(GRADE %in% c("A", "A-")) / n() * 100,
    Total = n(),

  ) %>%
  # Create numeric year for proper line connections
  mutate(Year_Num = as.numeric(substr(Academic_Year, 1, 4))) %>%
  arrange(Division, Year_Num)

# Get labels for x-axis
year_labels <- division_trends %>%
  select(Year_Num, Academic_Year) %>%
  distinct() %>%
  arrange(Year_Num)

  geom_point(size = 2) +
  scale_x_continuous(breaks = year_labels$Year_Num, labels = year_labels$Academic_Year) +
  scale_y_continuous(limits = c(0, 80), breaks = seq(0, 80, 10)) +
  scale_color_brewer(palette = "Set1") +
  labs(
    title = "A/A- Rate by Division Over Time",
    x = "Academic Year",
    y = "A/A- Percentage",
    color = "Division"
  ) +
  theme_minimal(base_size = 14) +
  theme(

dept_rates <- data %>%
  group_by(Department) %>%
